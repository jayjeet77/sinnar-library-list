function setupFilters() { const sel = document.getElementById('categoryFilter'); const nav = document.getElementById('categoryNav'); // १. तुमच्या मागणीनुसार अचूक क्रमवारी const customOrder = [ 'कादंबरी', 'इतिहास', 'कथा', 'कथासंग्रह', 'चरित्र', 'आत्मचरित्र', 'ललित', 'बालसाहित्य', 'कविता', 'नाटक', 'संदर्भ', 'राजकीय', 'शास्त्रीय', 'English', 'इतर' ]; const sheetCats = [...new Set(allBooks.map(b => b.Class))].filter(Boolean); let finalCats = []; // customOrder नुसार विभाग लावणे customOrder.forEach(orderCat => { if (sheetCats.includes(orderCat)) { finalCats.push(orderCat); } }); // जर शीटमध्ये असा एखादा विभाग असेल जो customOrder मध्ये नाही, तो 'English' च्या आधी जोडला जाईल sheetCats.forEach(sheetCat => { if (!finalCats.includes(sheetCat) && sheetCat !== 'English' && sheetCat !== 'इतर') { // English आणि इतर च्या मागे नवीन विभाग ढकलणे const insertIndex = finalCats.indexOf('English') !== -1 ? finalCats.indexOf('English') : finalCats.length; finalCats.splice(insertIndex, 0, sheetCat); } }); // २. नेव्हिगेशन आणि ड्रॉपडाऊनमध्ये विभाग जोडणे finalCats.forEach(c => { let o = document.createElement('option'); o.value = o.innerText = c; sel.appendChild(o); let a = document.createElement('a'); a.href = "javascript:void(0)"; a.innerText = c; a.onclick = function() { filterByCategory(c, this); }; nav.appendChild(a); }); }
